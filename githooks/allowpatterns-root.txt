# allowpatterns-root.txt — допуски (whitelist) КОРНЕВОГО хука (репо артефактов) + секрет-детектора
# клиентского pre-commit. Одна regex (grep -E) на строку. Пустые строки и '#'-комментарии игнор.
# Если строка, на которой сработал детектор секретов (SECRET_CONTENT_RE), дополнительно матчит
# любой из этих паттернов — срабатывание игнорируется (документация/пример, а не реальный секрет).
# Скан кейс-инсенситивный.
#
# --- строки-примеры в гайдах (начинаются с ❌ / ✅) ---
❌
✅
#
# --- инлайн-код в backticks (документация: `PASSWORD=`, `SECRET_KEY=` и т.п.) ---
`[^`]*`
#
# --- плейсхолдеры вместо реальных значений ---
sk-live-abc
replace-me
replace_me
change-?me
example
your[-_]?(token|secret|password|key|value)
<[^>]*>
\{\{[^}]*\}\}
\.\.\.
«[^»]*»
xxxx+
#
# --- заголовки диффа для зон, где документация про секреты легитимна ---
^\+\+\+ b/(ai/|ai/ops/security-rails\.md)
^--- a/(ai/|ai/ops/security-rails\.md)
^diff --git .*(ai/|ai/ops/security-rails\.md)
# документация детектора: цепочка ключевых слов через =/  (напр. =/SECRET_KEY=/TOKEN=) — не реальный секрет
=/(password|passwd|secret_key|secret|api_key|apikey|token|access_key)[[:space:]]*=

# --- хеши/SHA lock-файлов (composer.lock, package-lock.json): контент-адресуемые, не секреты ---
"(reference|content-hash|shasum|integrity)": *"[0-9a-f]{32,}"
